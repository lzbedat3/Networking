#!/usr/bin/env python3
"""
TCP Blocking Server - Python Implementation
Equivalent to TCP_BlockingServer.c

This server demonstrates:
1. Creating a TCP socket
2. Binding to a local port
3. Listening for incoming connections
4. Accepting client connections
5. Handling multiple client requests
6. Processing different types of requests
7. Running in an infinite loop to serve multiple clients
"""

import socket
import sys
from datetime import datetime

TIME_PORT = 27015  # Server port


def check_for_error(bytes_result, error_at, socket_1, socket_2):
    """
    Check if there was an error and handle it accordingly

    Args:
        bytes_result: Result from send/recv function
        error_at: String stating if it was receive or send function
        socket_1: First socket to close in case of error
        socket_2: Second socket to close in case of error

    Returns:
        True if error occurred, False otherwise
    """
    if bytes_result <= 0:
        print(f"Time Server: Error at {error_at}()")
        if socket_1:
            socket_1.close()
        if socket_2:
            socket_2.close()
        return True
    return False


def main():
    """
    Server Side Steps:
    1. Create Socket
    2. Bind to address and port
    3. Listen for connections
    4. Accept connections
    5. Send and Receive data
    """

    # Create a TCP socket
    # AF_INET = IPv4, SOCK_STREAM = TCP, IPPROTO_TCP = TCP protocol
    try:
        listen_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM, socket.IPPROTO_TCP)
        print("TCP socket created successfully")
    except socket.error as e:
        print(f"Time Server: Error at socket(): {e}")
        sys.exit(1)

    # Construct local address structure
    # '' or '0.0.0.0' means INADDR_ANY - accept connections on all interfaces
    server_service = ('', TIME_PORT)

    # Bind to the local address
    # The bind function establishes a connection to a specified socket
    try:
        listen_socket.bind(server_service)
        print(f"Socket bound to port {TIME_PORT}")
    except socket.error as e:
        print(f"Time Server: Error at bind(): {e}")
        listen_socket.close()
        sys.exit(1)

    # Listen on the socket for incoming connections
    # Backlog parameter = 5 (max length of pending connections queue)
    try:
        listen_socket.listen(5)
        print(f"Socket listening with backlog of 5")
    except socket.error as e:
        print(f"Time Server: Error at listen(): {e}")
        listen_socket.close()
        sys.exit(1)

    print(f"Time Server: Ready to accept connections on port {TIME_PORT}")

    # Main server loop - handle clients one by one (blocking)
    try:
        while True:
            print("Time Server: Wait for clients' requests.")

            # Accept an incoming connection
            # This call blocks until a client connects
            # Returns a new socket for communication with the client
            try:
                msg_socket, from_addr = listen_socket.accept()
                print(f"Time Server: Client is connected.")
                print(f"Client address: {from_addr[0]}:{from_addr[1]}")
            except socket.error as e:
                print(f"Time Server: Error at accept(): {e}")
                listen_socket.close()
                return

            # Handle this client's requests
            while True:
                # Receive data from the client
                try:
                    recv_buff = msg_socket.recv(255)
                    bytes_recv = len(recv_buff)

                    if check_for_error(bytes_recv, "recv", listen_socket, msg_socket):
                        return

                    if bytes_recv == 0:
                        print("Time Server: Client disconnected.")
                        msg_socket.close()
                        break

                    # Decode received message
                    recv_message = recv_buff.decode()
                    print(f"Received: {recv_message}")

                except socket.error as e:
                    print(f"Time Server: Error at recv(): {e}")
                    msg_socket.close()
                    break

                # Process the client request
                if recv_message.startswith("What's the time?"):
                    # Get current time
                    current_time = datetime.now()
                    send_buff = current_time.strftime("%Y-%m-%d %H:%M:%S")

                elif recv_message.startswith("Throw a random word."):
                    send_buff = "A Random Word :)"

                elif recv_message.startswith("What's your name?"):
                    send_buff = "My Name is Time Server, Nice to meet you."

                else:
                    # Any other message means exit
                    print("Time Server: Closing Connection.")
                    msg_socket.close()
                    break

                # Send response to client
                try:
                    print(f"Sending: {send_buff}")
                    bytes_sent = msg_socket.send(send_buff.encode())

                    if check_for_error(bytes_sent, "send", listen_socket, msg_socket):
                        return

                except socket.error as e:
                    print(f"Time Server: Error at send(): {e}")
                    msg_socket.close()
                    break

    except KeyboardInterrupt:
        print("\nTime Server: Shutting down...")
    finally:
        # Close the listening socket
        listen_socket.close()
        print("Socket closed")


if __name__ == "__main__":
    main()
