server
sock = socket(AF_INET, SOCK_DGRAM)
sock.bind(('', 5000))
while True:
 data, addr = sock.recvfrom(255)
 sock.sendto(now().encode(), addr)
client
SERVER = ("127.0.0.1", 5000)
sock.sendto(b"What's the time?", SERVER)
data, addr = sock.recvfrom(255)
sock.close()
TCP in Python
server = socket(AF_INET, SOCK_STREAM)
server.bind(('', 27015)); server.listen(5)
conn, addr = server.accept() # new socket
conn.send(answer(conn.recv(255)))
sock = socket(AF_INET, SOCK_STREAM)
sock.connect(("127.0.0.1", 27015)) # handshake
sock.send(b"What's the time?"); sock.recv(255)
 accept() returns a new socket per client
